\section*{A sample objective}
Assume $n\geq 1$ and $\sum_{i=1}^n x_i^2>0$.
\[
\mathcal{L}(\theta)=\frac{1}{n}\sum_{i=1}^{n}(y_i-\theta x_i)^2.
\]
\[
\widehat{\theta}=
\frac{\sum_{i=1}^{n}x_i y_i}{\sum_{i=1}^{n}x_i^2}.
\]
